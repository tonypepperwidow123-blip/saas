  id uuid primary key default uuid_generate_v4(),

  name text not null,
  slug text unique not null,
  description text,
  short_desc text,
  category text,
  tags text[],
  current_version text,
  status text not null default 'pending' check (status in ('pending', 'approved', 'rejected', 'suspended')),
  rejection_note text,
  price numeric(10,2) not null default 0 check (price >= 0),
  thumbnail_url text,
  download_count integer not null default 0,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()

create table if not exists public.plugin_versions (
  id uuid primary key default uuid_generate_v4(),
  plugin_id uuid not null references public.plugins(id) on delete cascade,
  version text not null,
  zip_path text not null,
  changelog text,
  is_latest boolean not null default false,
  created_at timestamptz not null default now(),
  unique(plugin_id, version)
);

create table if not exists public.licenses (
  id uuid primary key default uuid_generate_v4(),
  plugin_id uuid not null references public.plugins(id) on delete restrict,
  customer_id uuid not null references public.profiles(id) on delete restrict,
  license_key text unique not null,
  status text not null default 'active' check (status in ('active', 'suspended', 'expired', 'revoked')),
  activation_limit integer not null default 1 check (activation_limit > 0),
  expiry_date timestamptz,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists public.activations (
  id uuid primary key default uuid_generate_v4(),
  license_id uuid not null references public.licenses(id) on delete cascade,
  site_url text not null,
  plugin_version text,
  is_active boolean not null default true,
  activated_at timestamptz not null default now(),
  deactivated_at timestamptz,
  unique(license_id, site_url)

create table if not exists public.orders (

-- Auto-create profile on signup
create or replace function public.handle_new_user()
returns trigger as $$
begin
  insert into public.profiles (id, name, email, role)
  values (
    new.id,
    coalesce(new.raw_user_meta_data->>'name', 'User'),
    new.email,
    coalesce(new.raw_user_meta_data->>'role', 'customer')
  );
  return new;
end;
$$ language plpgsql security definer;

drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created
  after insert on auth.users
  for each row execute procedure public.handle_new_user();

-- Update timestamps
create or replace function public.update_updated_at()
returns trigger as $$
begin
  new.updated_at = now();
  return new;
end;
$$ language plpgsql;

drop trigger if exists update_profiles_updated_at on public.profiles;
create trigger update_profiles_updated_at before update on public.profiles
  for each row execute function public.update_updated_at();

-- Disable RLS (using server-side auth via service role key)
alter table public.profiles disable row level security;
alter table public.plugins disable row level security;
alter table public.plugin_versions disable row level security;
alter table public.licenses disable row level security;
alter table public.activations disable row level security;
alter table public.orders disable row level security;

WordPress Plugin API

🔌
What this is Your backend exposes a set of WordPress-compatible API endpoints under /api/wp/. You embed these into your plugin's PHP code so WordPress can automatically check for updates, activate licenses, and download new versions.

Live API Endpoints

GET /api/wp/check-update Check if a new version is available
GET /api/wp/download Download the latest plugin ZIP (token-protected)
POST /api/wp/activate Activate a license on a WordPress site
POST /api/wp/deactivate Deactivate a license from a site
GET /api/wp/validate-license Validate a license key status

PHP — Check for Updates (add to your plugin file)

php · your-plugin.php
<?php
// ── PluginVault Update Checker ──────────────────────────
define( 'PLUGINVAULT_API',    'https://YOUR_RENDER_NAME.onrender.com/api/wp' );
define( 'PLUGINVAULT_SLUG',   'your-plugin-slug' );   // must match what's in the dashboard
define( 'PLUGINVAULT_LICENSE', get_option( 'your_plugin_license_key', '' ) );

add_filter( 'pre_set_site_transient_update_plugins', 'pluginvault_check_update' );
function pluginvault_check_update( $transient ) {
    if ( empty( $transient->checked ) ) return $transient;

    $response = wp_remote_get( add_query_arg( [
        'license_key'     => PLUGINVAULT_LICENSE,
        'plugin_slug'     => PLUGINVAULT_SLUG,
        'current_version' => $transient->checked[ plugin_basename( __FILE__ ) ] ?? '1.0.0',
    ], PLUGINVAULT_API . '/check-update' ) );

    if ( is_wp_error( $response ) ) return $transient;

    $body = json_decode( wp_remote_retrieve_body( $response ), true );

    if ( ! empty( $body['data']['update_available'] ) ) {
        $transient->response[ plugin_basename( __FILE__ ) ] = (object) [
            'slug'        => PLUGINVAULT_SLUG,
            'new_version' => $body['data']['new_version'],
            'package'     => $body['data']['download_url'],
            'url'         => 'https://your-app.vercel.app',
        ];
    }

    return $transient;
}
?>

PHP — License Activation (settings page)

php · license-activation.php
<?php
function pluginvault_activate_license( $license_key, $activation_code ) {
    $response = wp_remote_post( PLUGINVAULT_API . '/activate', [
        'headers' => [ 'Content-Type' => 'application/json' ],
        'body'    => wp_json_encode( [
            'license_key'     => $license_key,
            'activation_code' => $activation_code,
            'site_url'        => home_url(),
            'plugin_slug'     => PLUGINVAULT_SLUG,
            'plugin_version'  => YOUR_PLUGIN_VERSION,
        ] ),
    ] );

    if ( is_wp_error( $response ) ) {
        return [ 'success' => false, 'message' => $response->get_error_message() ];
    }

    $body = json_decode( wp_remote_retrieve_body( $response ), true );

    if ( $body['success'] ) {
        update_option( 'your_plugin_license_key', $license_key );
        update_option( 'your_plugin_license_status', 'active' );
    }

    return $body;
}

function pluginvault_deactivate_license( $license_key ) {
    $response = wp_remote_post( PLUGINVAULT_API . '/deactivate', [
        'headers' => [ 'Content-Type' => 'application/json' ],
        'body'    => wp_json_encode( [
            'license_key' => $license_key,
            'site_url'    => home_url(),
        ] ),
    ] );

    $body = json_decode( wp_remote_retrieve_body( $response ), true );

    if ( $body['success'] ) {
        delete_option( 'your_plugin_license_key' );
        delete_option( 'your_plugin_license_status' );
    }

    return $body;
}
?>

PHP — Validate License on Every Page Load

php
<?php
function pluginvault_is_licensed() {
    $license_key = get_option( 'your_plugin_license_key', '' );
    if ( empty( $license_key ) ) return false;

    // Cache result for 12 hours to avoid hammering the API
    $cached = get_transient( 'pluginvault_license_valid' );
    if ( $cached !== false ) return (bool) $cached;

    $response = wp_remote_get( add_query_arg( [
        'license_key' => $license_key,
        'site_url'    => home_url(),
        'plugin_slug' => PLUGINVAULT_SLUG,
    ], PLUGINVAULT_API . '/validate-license' ) );

    $body = json_decode( wp_remote_retrieve_body( $response ), true );
    $valid = ! empty( $body['data']['is_valid'] );

    set_transient( 'pluginvault_license_valid', $valid, 12 * HOUR_IN_SECONDS );
    return $valid;
}

// Usage in your plugin:
// if ( ! pluginvault_is_licensed() ) {
//     // Show "please activate your license" notice
//     // and block premium features
// }
?>
ℹ️
How the activation flow works 1. Customer buys a plugin → a license key + one-time activation code are generated in the dashboard.
2. Customer copies both into your plugin's settings page.
3. Your PHP calls /api/wp/activate → the code is verified and consumed (one-use only).
4. From then on, /api/wp/validate-license keeps the site authorized.

Common Errors & Fixes

API calls fail with "Network Error"

🔴
Check both URLs match exactly (no trailing slash) VITE_API_URL on Vercel must end in /api
CLIENT_URL on Render must be your exact Vercel origin
Redeploy both services after changing env vars.

Blank page on refresh (e.g. /dashboard)

ℹ️
Already fixed by frontend/vercel.json. Make sure that file is committed to GitHub and not in .gitignore.

Render is very slow on first request (30–60s)

⚠️
Free tier services sleep after 15 min of inactivity. Use UptimeRobot (free) to ping your health URL every 10 minutes:
uptime monitor url
https://YOUR_RENDER_NAME.onrender.com/api/health

Emails not sending

⚠️
RESEND_API_KEY is empty on Render. Get a free key at resend.com → add it → redeploy.

WordPress "Invalid activation code"

ℹ️
Activation codes are one-time use. Once consumed, the same code cannot be reused. The customer needs to contact you for a new one, or you can reset it in the Supabase licenses table by setting activation_code_used = false.

VITE_ env vars not working on frontend

🔴
All Vite env vars must start with VITE_ — otherwise Vite ignores them. After adding/changing env vars on Vercel, trigger a manual redeploy.
